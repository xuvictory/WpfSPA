> [!example] 完整示例
> **自定义拖拽数据格式演示：定义私有拖拽格式字符串，把整个设备对象（不只是文本）封装进 DataObject，接收方用 SetData / GetData 按自定义格式读写复杂对象：**
>
> **MainWindow.xaml：**
> ```xml
> <Window x:Class="HmiDemo.MainWindow"
>         xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"
>         xmlns:x="http://schemas.microsoft.com/winfx/2006/xaml"
>         Title="自定义拖拽数据格式" Height="420" Width="520"
>         WindowStartupLocation="CenterScreen" Background="#0D1117">
>     <Grid Margin="15">
>         <Grid.RowDefinitions>
>             <RowDefinition Height="Auto"/>
>             <RowDefinition Height="*"/>
>         </Grid.RowDefinitions>
>         <TextBlock Text="自定义 DataObject 格式拖拽设备对象（保留全部字段）"
>                    Foreground="#58A6FF" FontSize="14" FontWeight="Bold" TextWrapping="Wrap"/>
>         <Grid Grid.Row="1" Margin="0,12,0,0">
>             <Grid.ColumnDefinitions>
>                 <ColumnDefinition Width="*"/>
>                 <ColumnDefinition Width="*"/>
>             </Grid.ColumnDefinitions>
>             <ListBox x:Name="DeviceList" Background="#161B22" BorderBrush="#21262D"
>                      PreviewMouseLeftButtonDown="OnDeviceMouseDown"
>                      PreviewMouseMove="OnDeviceMouseMove" Margin="0,0,6,0"/>
>             <Border Grid.Column="1" Background="#161B22" BorderBrush="#21262D"
>                     BorderThickness="1" CornerRadius="4" Margin="6,0,0,0" AllowDrop="True"
>                     DragEnter="OnDragEnter" Drop="OnDrop">
>                 <StackPanel Margin="10">
>                     <TextBlock Text="设备明细（自定义格式拖入）" Foreground="#58A6FF"
>                                FontWeight="Bold" HorizontalAlignment="Center"/>
>                     <StackPanel x:Name="DetailPanel" Margin="0,10,0,0">
>                         <TextBlock Text="← 拖一台设备过来查看完整信息" Foreground="#8B949E"
>                                    TextWrapping="Wrap"/>
>                     </StackPanel>
>                 </StackPanel>
>             </Border>
>         </Grid>
>     </Grid>
> </Window>
> ```
>
> **MainWindow.xaml.cs —— 后台代码：**
> ```csharp
> using System;
> using System.Collections.Generic;
> using System.Windows;
> using System.Windows.Controls;
> using System.Windows.Input;
> using System.Windows.Media;
>
> namespace HmiDemo
> {
>     // 设备对象：拖拽时整体传递，接收方拿到的是完整对象而非字符串
>     public class Device
>     {
>         public string Name { get; set; }
>         public string Type { get; set; }
>         public int Status { get; set; } // 0=停止 1=运行
>         public override string ToString() => Name;
>     }
>
>     public partial class MainWindow : Window
>     {
>         // 自定义拖拽格式：用 GUID 保证全局唯一，避免与其他应用冲突
>         public static readonly string DeviceFormat = "HmiDemo.DeviceFormat.3F2A1B4C";
>         private Point _dragStart;
>
>         public MainWindow()
>         {
>             InitializeComponent();
>             DeviceList.ItemsSource = new List<Device>
>             {
>                 new Device { Name = "1# 注塑机", Type = "注塑", Status = 1 },
>                 new Device { Name = "2# 注塑机", Type = "注塑", Status = 0 },
>                 new Device { Name = "3# 干燥机", Type = "干燥", Status = 1 },
>             };
>         }
>
>         private void OnDeviceMouseDown(object sender, MouseButtonEventArgs e)
>         {
>             _dragStart = e.GetPosition(null);
>         }
>
>         // 发起拖拽：SetData 使用自定义格式写入整个对象
>         private void OnDeviceMouseMove(object sender, MouseEventArgs e)
>         {
>             if (e.LeftButton != MouseButtonState.Pressed) return;
>             var pos = e.GetPosition(null);
>             if (Math.Abs(pos.X - _dragStart.X) < SystemParameters.MinimumHorizontalDragDistance &&
>                 Math.Abs(pos.Y - _dragStart.Y) < SystemParameters.MinimumVerticalDragDistance) return;
>
>             if (DeviceList.SelectedItem is Device device)
>             {
>                 var data = new DataObject();
>                 data.SetData(DeviceFormat, device); // 自定义格式装载对象
>                 data.SetData(DataFormats.Text, device.Name); // 顺带一份文本，便于外部程序读取
>                 DragDrop.DoDragDrop(DeviceList, data, DragDropEffects.Copy);
>             }
>         }
>
>         // 校验拖入数据是否带自定义格式
>         private void OnDragEnter(object sender, DragEventArgs e)
>         {
>             e.Effects = e.Data.GetDataPresent(DeviceFormat)
>                 ? DragDropEffects.Copy : DragDropEffects.None;
>             e.Handled = true;
>         }
>
>         // 落点：GetData 按自定义格式还原完整对象
>         private void OnDrop(object sender, DragEventArgs e)
>         {
>             DetailPanel.Children.Clear();
>             if (e.Data.GetData(DeviceFormat) is Device device)
>             {
>                 DetailPanel.Children.Add(NewText($"设备名称：{device.Name}", "White", true));
>                 DetailPanel.Children.Add(NewText($"设备类型：{device.Type}", "#8B949E", false));
>                 DetailPanel.Children.Add(NewText($"运行状态：{(device.Status == 1 ? "运行中" : "已停止")}",
>                     device.Status == 1 ? "#238636" : "#DA3633", false));
>             }
>             e.Handled = true;
>         }
>
>         // 快捷生成一行文本，用于动态展示拖入的设备信息
>         private static TextBlock NewText(string text, string color, bool bold)
>             => new TextBlock
>             {
>                 Text = text,
>                 Foreground = new SolidColorBrush((Color)ColorConverter.ConvertFromString(color)),
>                 FontWeight = bold ? FontWeights.Bold : FontWeights.Normal,
>                 Margin = new Thickness(0, 2, 0, 2)
>             };
>     }
> }
> ```

> - 经常重构：写完功能后回头看看有没有更简洁的写法

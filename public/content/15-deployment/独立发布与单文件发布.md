> [!example] 完整示例
> **发布环境诊断面板：Environment.Version 读取运行时版本、Assembly.Location 判断单文件发布、发布目录文件统计：**
>
> **MainWindow.xaml：**
> ```xml
> <Window x:Class="HmiDemo.MainWindow"
>         xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"
>         xmlns:x="http://schemas.microsoft.com/winfx/2006/xaml"
>         Title="发布环境诊断 - 独立/单文件发布" Height="430" Width="520"
>         WindowStartupLocation="CenterScreen" Background="#0D1117">
>     <StackPanel Margin="15">
>         <TextBlock Text="发布环境诊断" FontSize="18" FontWeight="Bold"
>                    Foreground="#58A6FF" Margin="0,0,0,10"/>
>         <Border Background="#161B22" Padding="12" CornerRadius="6" Margin="0,0,0,10">
>             <StackPanel>
>                 <TextBlock x:Name="TxtRuntime" Foreground="#8B949E" Margin="0,2"/>
>                 <TextBlock x:Name="TxtLocation" Foreground="#8B949E" Margin="0,2" TextWrapping="Wrap"/>
>                 <TextBlock x:Name="TxtSingleFile" Foreground="#8B949E" Margin="0,2"/>
>                 <TextBlock x:Name="TxtFileCount" Foreground="#8B949E" Margin="0,2"/>
>             </StackPanel>
>         </Border>
>         <Button Content="重新检测" Click="OnCheckClick" Padding="8" Margin="0,0,0,10"
>                 Background="#21262D" Foreground="White"/>
>         <TextBlock x:Name="TxtCommand" Foreground="#8B949E" TextWrapping="Wrap"
>                    FontFamily="Consolas" FontSize="12"/>
>     </StackPanel>
> </Window>
> ```
>
> **MainWindow.xaml.cs —— 后台代码：**
> ```csharp
> using System;
> using System.IO;
> using System.Reflection;
> using System.Windows;
>
> namespace HmiDemo
> {
>     public partial class MainWindow : Window
>     {
>         public MainWindow()
>         {
>             InitializeComponent();
>             Loaded += (_, _) => OnCheckClick(null, null);   // 启动即检测一次
>         }
>
>         private void OnCheckClick(object sender, RoutedEventArgs e)
>         {
>             try
>             {
>                 // Environment.Version：获取当前 .NET 运行时版本（独立发布时为随包携带的运行时版本）
>                 TxtRuntime.Text = "运行时版本：" + Environment.Version;
>                 // Assembly.Location：单文件发布时返回空字符串，可据此判断发布形态
>                 bool isSingleFile = string.IsNullOrEmpty(Assembly.GetEntryAssembly()?.Location);
>                 TxtLocation.Text = "程序集路径：" + (isSingleFile ? "（单文件/内存加载）"
>                     : Assembly.GetEntryAssembly().Location);
>                 // AppContext.BaseDirectory：运行时解包后的基础目录
>                 TxtSingleFile.Text = "单文件发布：" + (isSingleFile ? "是（所有依赖已打包进单个 exe）" : "否");
>                 // 统计基础目录文件数：单文件发布时通常只剩 1-2 个文件
>                 TxtFileCount.Text = "发布目录文件数：" + Directory.GetFiles(AppContext.BaseDirectory).Length;
>                 TxtCommand.Text = "参考发布命令（框架依赖）：" + Environment.NewLine +
>                     "dotnet publish -c Release -r win-x64" + Environment.NewLine +
>                     "参考发布命令（单文件）：" + Environment.NewLine +
>                     "dotnet publish -c Release -r win-x64 --self-contained /p:PublishSingleFile=true";
>             }
>             catch (Exception ex)
>             {
>                 MessageBox.Show("检测失败：" + ex.Message, "发布环境诊断");
>             }
>         }
>     }
> }
> ```
>

> [!example] 完整示例
> **设备参数配置面板：读取随发布目录分发的 appsettings.json（PLC IP/端口、串口、采集周期）并回显，配置缺失时使用默认值兜底：**
>
> **MainWindow.xaml：**
> ```xml
> <Window x:Class="HmiDemo.MainWindow"
>         xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"
>         xmlns:x="http://schemas.microsoft.com/winfx/2006/xaml"
>         Title="依赖项打包与配置文件" Height="460" Width="580"
>         WindowStartupLocation="CenterScreen" Background="#0D1117">
>     <StackPanel Margin="15">
>         <TextBlock Text="设备参数配置（appsettings.json）" FontSize="18" FontWeight="Bold"
>                    Foreground="#58A6FF" Margin="0,0,0,10"/>
>         <Border Background="#161B22" Padding="12" CornerRadius="6" Margin="0,0,0,10">
>             <StackPanel>
>                 <TextBlock x:Name="TxtPlcIp" Foreground="#8B949E" Margin="0,2"/>
>                 <TextBlock x:Name="TxtPlcPort" Foreground="#8B949E" Margin="0,2"/>
>                 <TextBlock x:Name="TxtSerialPort" Foreground="#8B949E" Margin="0,2"/>
>                 <TextBlock x:Name="TxtBaudRate" Foreground="#8B949E" Margin="0,2"/>
>                 <TextBlock x:Name="TxtPollInterval" Foreground="#8B949E" Margin="0,2"/>
>             </StackPanel>
>         </Border>
>         <Button Content="重新读取配置文件" Click="OnLoadConfig" Padding="8" Margin="0,0,0,10"
>                 Background="#21262D" Foreground="White"/>
>         <TextBlock x:Name="TxtJson" Foreground="#8B949E" TextWrapping="Wrap"
>                    FontFamily="Consolas" FontSize="12"/>
>     </StackPanel>
> </Window>
> ```
>
> **MainWindow.xaml.cs —— 后台代码：**
> ```csharp
> using System;
> using System.IO;
> using System.Text.Json;
> using System.Windows;
>
> namespace HmiDemo
> {
>     // 与 appsettings.json 对应的数据结构（发布时随程序一起拷贝到输出目录）
>     public class AppConfig
>     {
>         public string PlcIp { get; set; } = "192.168.1.10";
>         public int PlcPort { get; set; } = 502;
>         public string SerialPort { get; set; } = "COM3";
>         public int BaudRate { get; set; } = 9600;
>         public int PollIntervalMs { get; set; } = 100;
>     }
>
>     public partial class MainWindow : Window
>     {
>         public MainWindow()
>         {
>             InitializeComponent();
>             Loaded += (_, _) => OnLoadConfig(null, null);
>         }
>
>         private void OnLoadConfig(object sender, RoutedEventArgs e)
>         {
>             try
>             {
>                 // 配置文件与 exe 同目录分发，使用相对路径读取，避免硬编码绝对路径
>                 string path = Path.Combine(AppContext.BaseDirectory, "appsettings.json");
>                 AppConfig cfg;
>                 if (File.Exists(path))
>                 {
>                     cfg = JsonSerializer.Deserialize<AppConfig>(File.ReadAllText(path)) ?? new AppConfig();
>                     TxtJson.Text = "已加载配置：" + Environment.NewLine + File.ReadAllText(path);
>                 }
>                 else
>                 {
>                     // 配置文件缺失时使用默认值兜底，程序照常运行
>                     cfg = new AppConfig();
>                     TxtJson.Text = "未找到 appsettings.json，使用默认配置：" + Environment.NewLine +
>                         "{\n  \"PlcIp\": \"192.168.1.10\",\n  \"PlcPort\": 502,\n" +
>                         "  \"SerialPort\": \"COM3\",\n  \"BaudRate\": 9600,\n  \"PollIntervalMs\": 100\n}";
>                 }
>
>                 TxtPlcIp.Text = "PLC IP：" + cfg.PlcIp;
>                 TxtPlcPort.Text = "PLC 端口：" + cfg.PlcPort;
>                 TxtSerialPort.Text = "串口号：" + cfg.SerialPort;
>                 TxtBaudRate.Text = "波特率：" + cfg.BaudRate;
>                 TxtPollInterval.Text = "采集周期：" + cfg.PollIntervalMs + " ms";
>             }
>             catch (Exception ex)
>             {
>                 MessageBox.Show("读取配置失败：" + ex.Message, "配置文件");
>             }
>         }
>     }
> }
> ```
>

> 
> 坑 2：**忽略了官方文档** → Microsoft Docs 上有最权威的说明和最完整的示例代码，遇到问题先查文档

> [!best] 最佳实践
> - 编写代码时保持一致的命名规范（PascalCase 用于公共成员，_camelCase 用于私有字段）
> - 善用 Visual Studio 的智能提示和代码片段，提高开发效率
> - 每个关键代码块加上注释，解释"为什么这样写"而不仅仅是"写的是什么"
> - 遵循 SOLID 原则，尤其是单一职责原则：一个类只做一件事
> - 经常重构：写完功能后回头看看有没有更简洁的写法

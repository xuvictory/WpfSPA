> [!example] 完整示例
> **综合动画看板演示：将 Storyboard + 关键帧 + 缓动函数组合运用，模拟上位机实时监控页面的报警闪烁、趋势滚动、物料流动：**
>
> **MainWindow.xaml：**
> ```xml
> <Window x:Class="HmiDemo.MainWindow"
>         xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"
>         xmlns:x="http://schemas.microsoft.com/winfx/2006/xaml"
>         Title="动画看板 - 综合应用" Height="460" Width="520"
>         WindowStartupLocation="CenterScreen" Background="#0D1117">
>     <Grid Margin="15">
>         <Grid.RowDefinitions>
>             <RowDefinition Height="Auto"/>
>             <RowDefinition Height="*"/>
>             <RowDefinition Height="Auto"/>
>         </Grid.RowDefinitions>
>         <TextBlock Text="监控看板（综合动画）" Foreground="#58A6FF" FontSize="16" FontWeight="Bold"/>
>         <Grid Grid.Row="1" Margin="0,10,0,0" Background="#161B22">
>             <!-- 报警灯：颜色关键帧闪烁 -->
>             <Ellipse x:Name="AlarmBlink" Width="36" Height="36" Fill="#21262D"
>                      HorizontalAlignment="Left" VerticalAlignment="Top" Margin="20,20,0,0"/>
>             <!-- 流动物料：Storyboard 平移 -->
>             <Rectangle x:Name="FlowBox" Width="30" Height="18" Fill="#58A6FF" RadiusX="4" RadiusY="4"
>                        HorizontalAlignment="Left" VerticalAlignment="Center" Margin="0,0,0,0">
>                 <Rectangle.RenderTransform>
>                     <TranslateTransform x:Name="FlowMove" X="0" Y="0"/>
>                 </Rectangle.RenderTransform>
>             </Rectangle>
>             <!-- 状态文字 -->
>             <TextBlock x:Name="StatusLabel" Text="系统正常" Foreground="#8B949E" FontSize="16"
>                        HorizontalAlignment="Right" VerticalAlignment="Top" Margin="0,20,20,0"/>
>         </Grid>
>         <Button Grid.Row="2" Content="模拟报警" Click="OnAlarm" Margin="0,12,0,0"
>                 Padding="8" Background="#21262D" Foreground="White" HorizontalAlignment="Left"/>
>     </Grid>
> </Window>
> ```
>
> **MainWindow.xaml.cs —— 后台代码：**
> ```csharp
> using System.Windows;
> using System.Windows.Media;
> using System.Windows.Media.Animation;
>
> namespace HmiDemo
> {
>     public partial class MainWindow : Window
>     {
>         private readonly Storyboard _board = new Storyboard();
>         private bool _alarm;
>
>         public MainWindow()
>         {
>             InitializeComponent();
>             BuildAnimations();
>             _board.Begin(this);
>         }
>
>         // 组合：物料往复平移 + 报警灯常亮（默认关）
>         private void BuildAnimations()
>         {
>             // 物料往复流动
>             var move = new DoubleAnimationUsingKeyFrames
>             {
>                 Duration = new Duration(System.TimeSpan.FromSeconds(2)),
>                 RepeatBehavior = RepeatBehavior.Forever,
>                 AutoReverse = true
>             };
>             move.KeyFrames.Add(new EasingDoubleKeyFrame(0, KeyTime.FromTimeSpan(System.TimeSpan.Zero)));
>             move.KeyFrames.Add(new EasingDoubleKeyFrame(360,
>                 KeyTime.FromTimeSpan(System.TimeSpan.FromSeconds(1)),
>                 new CubicEase { EasingMode = EasingMode.EaseInOut }));
>             Storyboard.SetTarget(move, FlowMove);
>             Storyboard.SetTargetProperty(move, new PropertyPath(TranslateTransform.XProperty));
>             _board.Children.Add(move);
>         }
>
>         private void OnAlarm(object sender, RoutedEventArgs e)
>         {
>             _alarm = !_alarm;
>             if (_alarm)
>             {
>                 // 报警：红/黑快速闪烁
>                 var blink = new ColorAnimation(
>                     Color.FromRgb(0x21, 0x26, 0x2D), Color.FromRgb(0xDA, 0x36, 0x33),
>                     new Duration(System.TimeSpan.FromMilliseconds(300)))
>                 {
>                     AutoReverse = true,
>                     RepeatBehavior = RepeatBehavior.Forever
>                 };
>                 AlarmBlink.Fill = new SolidColorBrush();
>                 AlarmBlink.Fill.BeginAnimation(SolidColorBrush.ColorProperty, blink);
>                 StatusLabel.Text = "报警：1 号线超温";
>                 StatusLabel.Foreground = Brushes.OrangeRed;
>             }
>             else
>             {
>                 AlarmBlink.Fill.BeginAnimation(SolidColorBrush.ColorProperty, null); // 停止动画
>                 AlarmBlink.Fill = new SolidColorBrush(Color.FromRgb(0x21, 0x26, 0x2D));
>                 StatusLabel.Text = "系统正常";
>                 StatusLabel.Foreground = Brushes.LimeGreen;
>             }
>         }
>     }
> }
> ```

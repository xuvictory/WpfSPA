> [!example] 完整示例
> **报警系统演示：定时采集温度，超过阈值自动产生报警记录，可手动确认消除报警：**

> **MainWindow.xaml：**
> ```xml
> <Window x:Class="HmiDemo.MainWindow"
>         xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"
>         xmlns:x="http://schemas.microsoft.com/winfx/2006/xaml"
>         Title="报警系统" Height="440" Width="520"

>     <Grid Background="#161B22" Margin="10">

>             <RowDefinition Height="Auto"/>
>             <RowDefinition Height="Auto"/>
>             <RowDefinition Height="*"/>
>             <RowDefinition Height="Auto"/>
>         </Grid.RowDefinitions>
>
>         <!-- 阈值设置与采集控制 -->
>         <StackPanel Grid.Row="0" Orientation="Horizontal" Margin="5">
>             <TextBlock Text="报警阈值:" Foreground="#8B949E" VerticalAlignment="Center"/>
>             <TextBox x:Name="LimitBox" Width="60" Height="26" VerticalContentAlignment="Center"
>                      Background="#0D1117" Foreground="#8B949E" BorderBrush="#21262D" Text="50"/>

> **MainWindow.xaml.cs —— 后台代码：**
> ```csharp
> using System;
> using System.Windows;
> using System.Windows.Threading;
>
> namespace HmiDemo
> {
>     public partial class MainWindow : Window
>     {
>         private readonly DispatcherTimer _timer;
>         private readonly Random _rnd = new Random();
>
>         public MainWindow()
>         {
>             InitializeComponent();
>             // 每 500ms 采集一次，50% 概率越限制造报警
>             _timer = new DispatcherTimer { Interval = TimeSpan.FromMilliseconds(500) };
>             _timer.Tick += OnTick;
>         }
>
>         private void OnTick(object sender, EventArgs e)
>         {
>             double temp = _rnd.Next(30, 80);                 // 模拟采集
>             ValueText.Text = $"当前温度：{temp} ℃";
>
>             // 越限判定：超过阈值产生一条报警记录
>             if (temp > double.Parse(LimitBox.Text.Trim()))
>             {
>                 AlarmList.Items.Insert(0,
>                     $"[{DateTime.Now:HH:mm:ss}] 温度超限！当前 {temp}℃ > 阈值 {LimitBox.Text}℃");
>                 ValueText.Foreground = System.Windows.Media.Brushes.Red;
>             }
>             else
>             {
>                 ValueText.Foreground = System.Windows.Media.Brushes.LimeGreen;
>             }
>         }
>
>         private void OnStartClick(object sender, RoutedEventArgs e)
>         {
>             _timer.Start();
>             StartBtn.IsEnabled = false;
>             StopBtn.IsEnabled = true;
>         }
>
>         private void OnStopClick(object sender, RoutedEventArgs e)
>         {
>             _timer.Stop();
>             StartBtn.IsEnabled = true;
>             StopBtn.IsEnabled = false;
>         }
>
>         // 确认报警：清除报警列表(报警处理闭环的最后一个环节)
>         private void OnAckClick(object sender, RoutedEventArgs e)
>         {
>             AlarmList.Items.Clear();
>             ValueText.Text = "当前温度：-- ℃";
>             ValueText.Foreground = System.Windows.Media.Brushes.LimeGreen;
>         }
>     }
> }
> ```
>

> [!best] 最佳实践
> - 编写代码时保持一致的命名规范（PascalCase 用于公共成员，_camelCase 用于私有字段）
> - 善用 Visual Studio 的智能提示和代码片段，提高开发效率
> - 每个关键代码块加上注释，解释"为什么这样写"而不仅仅是"写的是什么"
> - 遵循 SOLID 原则，尤其是单一职责原则：一个类只做一件事
> - 经常重构：写完功能后回头看看有没有更简洁的写法

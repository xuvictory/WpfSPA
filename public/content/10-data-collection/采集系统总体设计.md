> **MainWindow.xaml.cs —— 后台代码：**
> ```csharp
> using System;
> using System.Collections.Concurrent;
> using System.Windows;
> using System.Windows.Media;
> using System.Windows.Threading;
>
> namespace HmiDemo
> {
>     public partial class MainWindow : Window
>     {
>         // 缓冲队列：采集层与处理层解耦的中间地带
>         private readonly ConcurrentQueue<string> _buffer = new ConcurrentQueue<string>();
>         private readonly DispatcherTimer _acqTimer;   // 采集层定时器(模拟设备采集)
>         private readonly DispatcherTimer _procTimer;  // 处理层定时器(模拟数据处理)
>         private readonly Random _rnd = new Random();
>         private int _total;                           // 总采集条数
>
>         public MainWindow()
>         {
>             InitializeComponent();
>
>             // 采集层：每 200ms 向队列写入一条模拟数据
>             _acqTimer = new DispatcherTimer { Interval = TimeSpan.FromMilliseconds(200) };
>             _acqTimer.Tick += (s, e) =>
>             {
>                 _buffer.Enqueue($"[采集层] {DateTime.Now:HH:mm:ss.fff} 设备1 温度={_rnd.Next(20, 60)}℃");
>                 _total++;
>                 UpdateStat();
>             };
>
>             // 处理层：每 500ms 从队列取出一条数据显示(生产快于消费，可观察队列积压)
>             _procTimer = new DispatcherTimer { Interval = TimeSpan.FromMilliseconds(500) };
>             _procTimer.Tick += (s, e) =>
>             {
>                 if (_buffer.TryDequeue(out string item))
>                     DataList.Items.Add(item);
>                 UpdateStat();
>             };
>         }
>
>         private void OnStartClick(object sender, RoutedEventArgs e)
>         {
>             _acqTimer.Start();
>             _procTimer.Start();
>             StartBtn.IsEnabled = false;
>             StopBtn.IsEnabled = true;
>             StatusText.Text = "采集链路：运行中(采集 200ms/次，处理 500ms/次)";
>             StatusText.Foreground = Brushes.LimeGreen;
>         }
>
>         private void OnStopClick(object sender, RoutedEventArgs e)
>         {
>             _acqTimer.Stop();
>             _procTimer.Stop();
>             StartBtn.IsEnabled = true;
>             StopBtn.IsEnabled = false;
>             StatusText.Text = "采集链路：已停止";
>             StatusText.Foreground = Brushes.OrangeRed;
>         }
>
>         // 统计层：实时刷新采集量与队列积压量
>         private void UpdateStat()
>             => StatText.Text = $"已采集 {_total} 条 | 队列积压 {_buffer.Count} 条";
>     }
> }
> ```

> [!pitfall] 常见踩坑
> 坑 1：**概念理解不清就上手** → 建议先把本章节的前置知识点学完，理解基础原理后再动手写代码
> 
> 坑 2：**忽略了官方文档** → Microsoft Docs 上有最权威的说明和最完整的示例代码，遇到问题先查文档
>
> 坑 3：**代码写的太"一次性"** → 养成写可复用代码的习惯，以后项目中会反复用到这些知识

> [!best] 最佳实践
> - 编写代码时保持一致的命名规范（PascalCase 用于公共成员，_camelCase 用于私有字段）
> - 善用 Visual Studio 的智能提示和代码片段，提高开发效率
> - 每个关键代码块加上注释，解释"为什么这样写"而不仅仅是"写的是什么"
> - 遵循 SOLID 原则，尤其是单一职责原则：一个类只做一件事
> - 经常重构：写完功能后回头看看有没有更简洁的写法

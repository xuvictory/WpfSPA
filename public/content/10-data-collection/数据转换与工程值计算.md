> **MainWindow.xaml.cs —— 后台代码：**
> ```csharp
> using System;
> using System.Windows;
>
> namespace HmiDemo
> {
>     public partial class MainWindow : Window
>     {
>         public MainWindow() => InitializeComponent();
>
>         private void OnConvertClick(object sender, RoutedEventArgs e)
>         {
>             double raw = double.Parse(RawBox.Text.Trim());
>             double engMin = double.Parse(EngMinBox.Text.Trim());
>             double engMax = double.Parse(EngMaxBox.Text.Trim());
>
>             // 核心换算：原始值按比例线性映射到工程值区间
>             double eng = LinearScale(raw, 0, 27648, engMin, engMax);
>
>             ResultText.Text = $"工程值：{eng:F2} ℃";
>             LogList.Items.Insert(0, $"原始值 {raw} → 工程值 {eng:F2} ℃（量程 {engMin}~{engMax}）");
>         }
>
>         // 通用线性标度公式：把 [rawMin, rawMax] 区间映射到 [engMin, engMax]
>         private double LinearScale(double raw, double rawMin, double rawMax,
>                                    double engMin, double engMax)
>         {
>             if (rawMax == rawMin) return engMin;                 // 防除零
>             double ratio = (raw - rawMin) / (rawMax - rawMin);   // 归一化到 0~1
>             return ratio * (engMax - engMin) + engMin;           // 映射到工程值区间
>         }
>     }
> }
> ```

> **三种采集策略演示：定时轮询、事件驱动报警、订阅推送数据流：**

> **MainWindow.xaml.cs —— 后台代码：**
> ```csharp
> using System;
> using System.Windows;
> using System.Windows.Threading;
>
> namespace HmiDemo
> {
>     public partial class MainWindow : Window
>     {
>         // 策略三用的“发布/订阅”事件：订阅者无需关心数据从哪里来
>         public event EventHandler<double> DataPushed;
>
>         private readonly DispatcherTimer _pollTimer;  // 轮询定时器
>         private readonly DispatcherTimer _pushTimer;  // 订阅推送定时器
>         private readonly Random _rnd = new Random();
>
>         public MainWindow()
>         {
>             InitializeComponent();
>
>             // 轮询：定时器到点就主动读一次设备
>             _pollTimer = new DispatcherTimer { Interval = TimeSpan.FromMilliseconds(500) };
>             _pollTimer.Tick += (s, e) => Log("轮询", $"主动读取：温度 {_rnd.Next(20, 60)} ℃");
>
>             // 订阅：定时推送数据源
>             _pushTimer = new DispatcherTimer { Interval = TimeSpan.FromSeconds(1) };
>             _pushTimer.Tick += (s, e) => DataPushed?.Invoke(this, _rnd.Next(20, 60));
>
>             // 订阅者：DataPushed 一旦触发，这个回调就执行
>             DataPushed += (s, value) => Log("订阅", $"被动接收推送：温度 {value} ℃");
>         }
>
>         private void OnStartPoll(object sender, RoutedEventArgs e)
>         {
>             _pollTimer.Start();
>             PollBtn.IsEnabled = false;
>             StopPollBtn.IsEnabled = true;
>         }
>
>         private void OnStopPoll(object sender, RoutedEventArgs e)
>         {
>             _pollTimer.Stop();
>             PollBtn.IsEnabled = true;
>             StopPollBtn.IsEnabled = false;
>         }
>
>         // 事件驱动：设备超温时主动通知，而不是等主程序来查
>         private void OnFireAlarm(object sender, RoutedEventArgs e)
>             => Log("事件", $"设备超温报警触发，当前温度 {_rnd.Next(70, 90)} ℃");
>
>         private void OnToggleSubscribe(object sender, RoutedEventArgs e)
>         {
>             if (_pushTimer.IsEnabled)
>             {
>                 _pushTimer.Stop();
>                 SubBtn.Content = "订阅数据流(每1s推送)";
>             }
>             else
>             {
>                 _pushTimer.Start();
>                 SubBtn.Content = "取消订阅";
>             }
>         }
>
>         private void Log(string strategy, string msg)
>             => LogList.Items.Insert(0, $"[{strategy}] {DateTime.Now:HH:mm:ss}  {msg}");
>     }
> }
> ```

> [!pitfall] 常见踩坑
> 坑 1：**概念理解不清就上手** → 建议先把本章节的前置知识点学完，理解基础原理后再动手写代码
> 
> 坑 2：**忽略了官方文档** → Microsoft Docs 上有最权威的说明和最完整的示例代码，遇到问题先查文档
>
> 坑 3：**代码写的太"一次性"** → 养成写可复用代码的习惯，以后项目中会反复用到这些知识

> [!best] 最佳实践
> - 编写代码时保持一致的命名规范（PascalCase 用于公共成员，_camelCase 用于私有字段）
> - 善用 Visual Studio 的智能提示和代码片段，提高开发效率
> - 每个关键代码块加上注释，解释"为什么这样写"而不仅仅是"写的是什么"
> - 遵循 SOLID 原则，尤其是单一职责原则：一个类只做一件事
> - 经常重构：写完功能后回头看看有没有更简洁的写法

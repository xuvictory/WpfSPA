> [!example] 完整示例
> **仪表盘演示：XAML 自绘圆形仪表盘，指针随 Slider 输入旋转，实时显示工程值：**
>
> **MainWindow.xaml：**
> ```xml
> <Window x:Class="HmiDemo.MainWindow"
>         xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"
>         xmlns:x="http://schemas.microsoft.com/winfx/2006/xaml"
>         Title="仪表盘控件" Height="420" Width="420"
>         WindowStartupLocation="CenterScreen" Background="#0D1117">
>     <Grid Background="#161B22" Margin="10">
>         <Grid.RowDefinitions>
>             <RowDefinition Height="*"/>
>             <RowDefinition Height="Auto"/>
>             <RowDefinition Height="Auto"/>
>         </Grid.RowDefinitions>
>
>         <!-- 仪表盘：Canvas 上叠加刻度底盘与指针 -->
>         <Canvas Grid.Row="0" Width="260" Height="260" HorizontalAlignment="Center">
>             <!-- 表盘底 -->
>             <Ellipse Canvas.Left="20" Canvas.Top="20" Width="220" Height="220"
>                      Stroke="#21262D" StrokeThickness="10"/>
>             <!-- 表盘内衬 -->
>             <Ellipse Canvas.Left="32" Canvas.Top="32" Width="196" Height="196"
>                      Fill="#0D1117"/>
>             <!-- 指针：以底部中心为轴旋转 -->
>             <Rectangle x:Name="Needle" Canvas.Left="127" Canvas.Top="36" Width="6" Height="92"
>                        Fill="#58A6FF" RadiusX="3" RadiusY="3"
>                        RenderTransformOrigin="0.5,0.9">
>                 <Rectangle.RenderTransform>
>                     <RotateTransform x:Name="NeedleRotate" Angle="-120"/>
>                 </Rectangle.RenderTransform>
>             </Rectangle>
>             <!-- 轴心 -->
>             <Ellipse Canvas.Left="122" Canvas.Top="121" Width="16" Height="16" Fill="#58A6FF"/>
>             <!-- 数值显示 -->
>             <TextBlock x:Name="ValueText" Canvas.Left="80" Canvas.Top="160" Width="100"
>                        Text="50.0" FontSize="28" Foreground="White" TextAlignment="Center"/>
>             <TextBlock Canvas.Left="95" Canvas.Top="196" Text="℃" Foreground="#8B949E"/>
>         </Canvas>
>
>         <!-- 模拟量输入 -->
>         <Slider x:Name="ValueSlider" Grid.Row="1" Minimum="0" Maximum="100" Value="50"
>                 Margin="20,0" ValueChanged="OnValueChanged"/>
>         <TextBlock x:Name="RawText" Grid.Row="2" Margin="5" Foreground="#8B949E"
>                    Text="模拟量：50.0 (0~100)"/>
>     </Grid>
> </Window>
> ```
>
> **MainWindow.xaml.cs —— 后台代码：**
> ```csharp
> using System;
> using System.Windows;
> using System.Windows.Controls;
> using System.Windows.Media;
>
> namespace HmiDemo
> {
>     public partial class MainWindow : Window
>     {
>         // 仪表盘跨度：-120°(0值) 到 +120°(满量程)，共 240°
>         private const double MinAngle = -120;
>         private const double MaxAngle = 120;
>
>         public MainWindow() => InitializeComponent();
>
>         private void OnValueChanged(object sender, RoutedPropertyChangedEventArgs<double> e)
>         {
>             double value = ValueSlider.Value;
>             // 把 0~100 映射为 -120°~+120°
>             double angle = MinAngle + value / 100.0 * (MaxAngle - MinAngle);
>             NeedleRotate.Angle = angle;
>             ValueText.Text = value.ToString("F1");
>             RawText.Text = $"模拟量：{value:F1} (0~100)";
>         }
>     }
> }
> ```
>

> - 每个关键代码块加上注释，解释"为什么这样写"而不仅仅是"写的是什么"
> - 遵循 SOLID 原则，尤其是单一职责原则：一个类只做一件事
> - 经常重构：写完功能后回头看看有没有更简洁的写法

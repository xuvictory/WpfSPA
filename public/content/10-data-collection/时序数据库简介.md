> **MainWindow.xaml.cs —— 后台代码：**
> ```csharp
> using System;
> using System.Collections.Generic;
> using System.Linq;
> using System.Windows;
>
> namespace HmiDemo
> {
>     public partial class MainWindow : Window
>     {
>         // 内存中的“时序数据表”：每条记录 = 时间戳 + 数值
>         private readonly List<(DateTime Time, double Value)> _ts = new List<(DateTime, double)>();
>         private readonly Random _rnd = new Random();
>
>         public MainWindow() => InitializeComponent();
>
>         // 写入：批量追加带时间戳的数据，并按时间排序(时序库的基本操作)
>         private void OnWriteClick(object sender, RoutedEventArgs e)
>         {
>             DateTime now = DateTime.Now;
>             for (int i = 0; i < 120; i++)
>                 _ts.Add((now.AddSeconds(-i * 2), _rnd.Next(20, 60)));   // 每 2 秒一条
>             _ts.Sort((a, b) => a.Time.CompareTo(b.Time));
>             StatText.Text = $"已写入 {_ts.Count} 条";
>             ResultList.Items.Insert(0, $"写入 {_ts.Count} 条，最新值 {_ts[_ts.Count - 1].Value}℃");
>         }
>
>         // 查询：按时间范围过滤(时序查询最常见形态)
>         private void OnQueryClick(object sender, RoutedEventArgs e)
>         {
>             DateTime since = DateTime.Now.AddMinutes(-1);
>             var rows = _ts.Where(t => t.Time >= since).ToList();
>             ResultList.Items.Insert(0,
>                 $"时间范围查询(最近1分钟)命中 {rows.Count} 条，平均 {rows.Average(t => t.Value):F1}℃");
>         }
>
>         // 降采样：按 10 秒窗口聚合出平均值(压缩数据量，长周期趋势分析常用)
>         private void OnAggregateClick(object sender, RoutedEventArgs e)
>         {
>             var groups = _ts.GroupBy(t => t.Time.Ticks / TimeSpan.FromSeconds(10).Ticks);
>             ResultList.Items.Insert(0, $"降采样聚合(10秒窗口)得到 {groups.Count()} 个点：");
>             foreach (var g in groups.Take(5))
>                 ResultList.Items.Add($"  {new DateTime(g.Key * TimeSpan.FromSeconds(10).Ticks):HH:mm:ss}" +
>                                      $"  平均 {g.Average(t => t.Value):F1}℃");
>         }
>     }
> }
> ```

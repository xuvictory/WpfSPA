> **MainWindow.xaml.cs —— 后台代码：**
> ```csharp
> using System.Windows;
> using System.Windows.Media;
>
> namespace HmiDemo
> {
>     // 状态机核心：状态枚举 + 合法迁移表，把"能不能迁"收敛到一处管理
>     public enum DeviceState { Stopped, Running, Paused, Faulted }
>
>     public partial class MainWindow : Window
>     {
>         private DeviceState _state = DeviceState.Stopped;
>
>         public MainWindow() => InitializeComponent();
>
>         // 尝试迁移：合法则执行并刷新 UI，非法则拦截并提示
>         private bool TryTransition(DeviceState target, string action)
>         {
>             // 元组模式 switch：(目标状态, 当前状态) → 是否允许
>             bool allowed = (target, _state) switch
>             {
>                 (DeviceState.Running, DeviceState.Stopped) => true, // 启动
>                 (DeviceState.Paused, DeviceState.Running) => true,  // 暂停
>                 (DeviceState.Running, DeviceState.Paused) => true,  // 继续
>                 (DeviceState.Faulted, _) => true,                   // 任意状态可报故障
>                 (DeviceState.Stopped, DeviceState.Faulted) => true, // 故障复位
>                 _ => false
>             };
>             if (!allowed)
>             {
>                 LogText.Text = $"非法操作：{action}（{_state} → {target} 不允许）";
>                 LogText.Foreground = Brushes.OrangeRed;
>                 return false;
>             }
>             _state = target;
>             StateText.Text = target.ToString();
>             StateText.Foreground = target switch
>             {
>                 DeviceState.Running => Brushes.LimeGreen,
>                 DeviceState.Faulted => Brushes.OrangeRed,
>                 _ => new SolidColorBrush(Color.FromRgb(0x8B, 0x94, 0x9E))
>             };
>             LogText.Text = $"操作成功：{action} → 当前状态 {target}";
>             LogText.Foreground = new SolidColorBrush(Color.FromRgb(0x8B, 0x94, 0x9E));
>             return true;
>         }
>
>         private void OnStart(object s, RoutedEventArgs e) => TryTransition(DeviceState.Running, "启动");
>         private void OnPause(object s, RoutedEventArgs e) => TryTransition(DeviceState.Paused, "暂停");
>         private void OnResume(object s, RoutedEventArgs e) => TryTransition(DeviceState.Running, "继续");
>         private void OnFault(object s, RoutedEventArgs e) => TryTransition(DeviceState.Faulted, "故障");
>         private void OnReset(object s, RoutedEventArgs e) => TryTransition(DeviceState.Stopped, "复位");
>     }
> }
> ```

> [!pitfall] 常见踩坑
> 坑 1：**概念理解不清就上手** → 建议先把本章节的前置知识点学完，理解基础原理后再动手写代码
> 
> 坑 2：**忽略了官方文档** → Microsoft Docs 上有最权威的说明和最完整的示例代码，遇到问题先查文档
>
> 坑 3：**代码写的太"一次性"** → 养成写可复用代码的习惯，以后项目中会反复用到这些知识

> [!best] 最佳实践
> - 编写代码时保持一致的命名规范（PascalCase 用于公共成员，_camelCase 用于私有字段）
> - 善用 Visual Studio 的智能提示和代码片段，提高开发效率
> - 每个关键代码块加上注释，解释"为什么这样写"而不仅仅是"写的是什么"
> - 遵循 SOLID 原则，尤其是单一职责原则：一个类只做一件事
> - 经常重构：写完功能后回头看看有没有更简洁的写法

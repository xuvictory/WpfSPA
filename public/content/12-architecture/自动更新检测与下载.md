> [!example] 完整示例
> **自动更新流程演示：启动后定时检查版本接口（模拟 HTTP 请求），发现新版本后下载进度条滚动，下载完成提示重启安装：**
>
> **MainWindow.xaml：**
> ```xml
> <Window x:Class="HmiDemo.MainWindow"
>         xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"
>         xmlns:x="http://schemas.microsoft.com/winfx/2006/xaml"
>         Title="自动更新演示" Height="360" Width="440"
>         WindowStartupLocation="CenterScreen" Background="#0D1117">
>     <StackPanel Margin="15">
>         <TextBlock Text="自动更新：检测 → 下载 → 重启" Foreground="#58A6FF" FontWeight="Bold"/>
>         <Button Content="检查更新" Click="OnCheck" Margin="0,12,0,0" Padding="8"
>                 Background="#21262D" Foreground="White"/>
>         <ProgressBar x:Name="DownloadBar" Height="18" Margin="0,12,0,0"
>                      Minimum="0" Maximum="100" Foreground="#58A6FF" Background="#161B22"/>
>         <Button Content="开始下载更新" Click="OnDownload" Margin="0,10,0,0" Padding="8"
>                 Background="#238636" Foreground="White"/>
>         <TextBlock x:Name="OutputText" Margin="0,12,0,0" Foreground="#8B949E" TextWrapping="Wrap"/>
>     </StackPanel>
> </Window>
> ```
>
> **MainWindow.xaml.cs —— 后台代码：**
> ```csharp
> using System;
> using System.Threading;
> using System.Windows;
> using System.Windows.Media;
> using System.Windows.Threading;
>
> namespace HmiDemo
> {
>     public partial class MainWindow : Window
>     {
>         private readonly DispatcherTimer _timer = new DispatcherTimer();
>         private int _progress = 0;
>
>         public MainWindow()
>         {
>             InitializeComponent();
>             _timer.Interval = TimeSpan.FromMilliseconds(150);
>             _timer.Tick += OnTick;
>         }
>
>         private void OnCheck(object sender, RoutedEventArgs e)
>         {
>             // 真实场景：请求 http://update.hmi.com/version 比对版本号
>             OutputText.Text = "服务器返回：最新版本 2.1.0（本地 2.0.5）\n→ 发现更新，可点击下载";
>             OutputText.Foreground = new SolidColorBrush(Color.FromRgb(0x23, 0x86, 0x36));
>         }
>
>         private void OnDownload(object sender, RoutedEventArgs e)
>         {
>             _progress = 0;
>             DownloadBar.Value = 0;
>             _timer.Start(); // 定时器模拟下载进度
>         }
>
>         private void OnTick(object sender, EventArgs e)
>         {
>             _progress += 5;
>             DownloadBar.Value = _progress;
>             if (_progress >= 100)
>             {
>                 _timer.Stop();
>                 OutputText.Text = "下载完成，校验 MD5 通过\n请重启程序完成安装";
>                 OutputText.Foreground = new SolidColorBrush(Color.FromRgb(0x23, 0x86, 0x36));
>             }
>         }
>     }
> }
> ```
>

> - 善用 Visual Studio 的智能提示和代码片段，提高开发效率
> - 每个关键代码块加上注释，解释"为什么这样写"而不仅仅是"写的是什么"
> - 遵循 SOLID 原则，尤其是单一职责原则：一个类只做一件事
> - 经常重构：写完功能后回头看看有没有更简洁的写法

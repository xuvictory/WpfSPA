> [!example] 完整示例
> **配置加密与热更新演示：密码字段用 AES 简单加密后存储，界面上修改设备 IP 后点击保存立即生效（模拟运行时修改配置）：**
>
> **MainWindow.xaml：**
> ```xml
> <Window x:Class="HmiDemo.MainWindow"
>         xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"
>         xmlns:x="http://schemas.microsoft.com/winfx/2006/xaml"
>         Title="配置加密与热更新" Height="400" Width="440"
>         WindowStartupLocation="CenterScreen" Background="#0D1117">
>     <StackPanel Margin="15">
>         <TextBlock Text="配置加密 + 运行时修改" Foreground="#58A6FF" FontWeight="Bold"/>
>         <TextBox x:Name="IpBox" Margin="0,12,0,0" Padding="4"
>                  Background="#161B22" Foreground="#8B949E" Text="192.168.1.50"/>
>         <Button Content="加密保存密码（123456）" Click="OnEncrypt" Margin="0,10,0,0" Padding="8"
>                 Background="#21262D" Foreground="White"/>
>         <Button Content="应用新 IP（热更新）" Click="OnApply" Margin="0,8,0,0" Padding="8"
>                 Background="#238636" Foreground="White"/>
>         <TextBlock x:Name="OutputText" Margin="0,12,0,0" Foreground="#8B949E" TextWrapping="Wrap"/>
>     </StackPanel>
> </Window>
> ```
>
> **MainWindow.xaml.cs —— 后台代码：**
> ```csharp
> using System;
> using System.Security.Cryptography;
> using System.Text;
> using System.Windows;
> using System.Windows.Media;
>
> namespace HmiDemo
> {
>     public partial class MainWindow : Window
>     {
>         private string _currentIp = "192.168.1.50";
>         private string _encryptedPassword = "";
>
>         public MainWindow() => InitializeComponent();
>
>         // AES 加密：明文 → 密文（生产环境密钥应从环境变量/密钥服务获取）
>         private static string Encrypt(string plain)
>         {
>             using var aes = Aes.Create();
>             aes.Key = SHA256.HashData(Encoding.UTF8.GetBytes("Hmi-Demo-Key"));
>             aes.IV = new byte[16];
>             using var encryptor = aes.CreateEncryptor();
>             byte[] result = encryptor.TransformFinalBlock(Encoding.UTF8.GetBytes(plain), 0, plain.Length);
>             return Convert.ToBase64String(result);
>         }
>
>         private void OnEncrypt(object sender, RoutedEventArgs e)
>         {
>             _encryptedPassword = Encrypt("123456");
>             OutputText.Text = $"密码已加密保存（密文）：\n{_encryptedPassword}";
>             OutputText.Foreground = new SolidColorBrush(Color.FromRgb(0x23, 0x86, 0x36));
>         }
>
>         private void OnApply(object sender, RoutedEventArgs e)
>         {
>             // 运行时热更新：修改后立即生效，无需重启
>             _currentIp = IpBox.Text;
>             OutputText.Text = $"IP 已热更新为 {_currentIp}（无需重启程序）\n" +
>                               (string.IsNullOrEmpty(_encryptedPassword)
>                                   ? "提示：请先点击上方按钮保存加密密码"
>                                   : "密码密文已就绪，可在连接时解密使用");
>             OutputText.Foreground = new SolidColorBrush(Color.FromRgb(0x23, 0x86, 0x36));
>         }
>     }
> }
> ```

> [!example] 完整示例
> **一个界面同时演示四种选择控件的适用差异：ComboBox 省空间、ListBox 常驻多选、ListView 表格化：**
>
> **MainWindow.xaml：**
> ```xml
> <Window x:Class="HmiDemo.MainWindow"
>         xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"
>         xmlns:x="http://schemas.microsoft.com/winfx/2006/xaml"
>         Title="选择控件对比 - ComboBox/ListBox/ListView" Height="520" Width="760"
>         WindowStartupLocation="CenterScreen" Background="#0D1117">
>     <Grid Margin="12">
>         <Grid.ColumnDefinitions>
>             <ColumnDefinition Width="2*"/>
>             <ColumnDefinition Width="10"/>
>             <ColumnDefinition Width="3*"/>
>         </Grid.ColumnDefinitions>
>
>         <!-- 左栏：ComboBox 单选 + ListBox 多选 -->
>         <StackPanel Grid.Column="0">
>             <TextBlock Text="ComboBox：空间紧凑，适合下拉单选" Foreground="#8B949E"/>
>             <ComboBox x:Name="cmbDevice" DisplayMemberPath="Name" Margin="0,4,0,12"
>                       Background="#161B22" Foreground="White" Padding="4"/>
>
>             <TextBlock Text="ListBox：常驻展示，支持多选" Foreground="#8B949E"/>
>             <ListBox x:Name="lstDevice" SelectionMode="Extended" Height="180"
>                      DisplayMemberPath="Name" Background="#161B22" Foreground="White"/>
>         </StackPanel>
>
>         <!-- 右栏：ListView 表格 -->
>         <ListView x:Name="lvDevice" Grid.Column="2" Background="#161B22"
>                   Foreground="White" BorderBrush="#2A4A6C" BorderThickness="1">
>             <ListView.View>
>                 <GridView>
>                     <GridViewColumn Header="编号" Width="80" DisplayMemberBinding="{Binding Id}"/>
>                     <GridViewColumn Header="名称" Width="140" DisplayMemberBinding="{Binding Name}"/>
>                     <GridViewColumn Header="状态" Width="90" DisplayMemberBinding="{Binding Status}"/>
>                 </GridView>
>             </ListView.View>
>         </ListView>
>     </Grid>
> </Window>
> ```
>
> **MainWindow.xaml.cs —— 后台代码：**
> ```csharp
> using System.Collections.Generic;
> using System.Windows;
>
> namespace HmiDemo
> {
>     public partial class MainWindow : Window
>     {
>         public MainWindow()
>         {
>             InitializeComponent();
>
>             // 同一份数据源可以同时喂给三种控件
>             var devices = new List<Device>
>             {
>                 new Device { Id = "D1", Name = "电机 M-101", Status = "运行" },
>                 new Device { Id = "D2", Name = "变频器 V-202", Status = "运行" },
>                 new Device { Id = "D3", Name = "传感器 S-303", Status = "待机" }
>             };
>             cmbDevice.ItemsSource = devices;
>             lstDevice.ItemsSource = devices;
>             lvDevice.ItemsSource = devices;
>         }
>     }
>
>     public class Device
>     {
>         public string Id { get; set; }
>         public string Name { get; set; }
>         public string Status { get; set; }
>     }
> }
> ```
>

> **MainWindow.xaml.cs —— 后台代码：**
> ```csharp
> using System;
> using System.Windows;
> using System.Windows.Media;
> using System.Windows.Threading;
>
> namespace HmiDemo
> {
>     public partial class MainWindow : Window
>     {
>         private readonly DispatcherTimer _timer = new DispatcherTimer { Interval = TimeSpan.FromSeconds(2) };
>         private readonly Random _random = new Random();
>         private int _output;      // 累计产量
>         private int _alarmCount;  // 累计报警数
>
>         public MainWindow()
>         {
>             InitializeComponent();
>             _timer.Tick += OnTimerTick;
>             _timer.Start();
>         }
>
>         private void OnTimerTick(object sender, EventArgs e)
>         {
>             // 模拟产线数据源每 2 秒推送一批数据
>             _output += _random.Next(3, 9);
>             OutputText.Text = _output.ToString();
>             OeeText.Text = _random.Next(80, 98) + "%";
>             YieldText.Text = _random.Next(96, 100) + "%";
>
>             // 偶发报警，演示报警滚动列表
>             if (_random.Next(0, 5) == 0)
>             {
>                 _alarmCount++;
>                 AlarmText.Text = _alarmCount.ToString();
>                 AlarmText.Foreground = Brushes.OrangeRed;
>                 var time = DateTime.Now.ToString("HH:mm:ss");
>                 AlarmListText.Text = $"[{time}] 3 号工位温度过高，请检查冷却回路\n" + AlarmListText.Text;
>             }
>         }
>     }
> }
> ```

> - 善用 Visual Studio 的智能提示和代码片段，提高开发效率
> - 每个关键代码块加上注释，解释"为什么这样写"而不仅仅是"写的是什么"
> - 遵循 SOLID 原则，尤其是单一职责原则：一个类只做一件事
> - 经常重构：写完功能后回头看看有没有更简洁的写法

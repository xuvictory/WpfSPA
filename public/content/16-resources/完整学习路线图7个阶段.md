> using System.Runtime.CompilerServices;

>                 new Stage { Name = "阶段 6 项目实战", Progress = 100 },
>                 new Stage { Name = "阶段 7 性能优化与部署", Progress = 80 }
>             };
>             SummaryText.Text = "已完成 6.8 / 7 个阶段";
>             StatusText.Text = "综合成果：已能独立开发带实时曲线、PLC 通信与日志记录的上位机程序";
>         }
>     }
>
>     // 阶段数据模型：实现 INotifyPropertyChanged 供进度条绑定更新
>     public class Stage : INotifyPropertyChanged
>     {
>         public string Name { get; set; }
>
>         private int _progress;
>         public int Progress
>         {
>             get => _progress;
>             set
>             {
>                 _progress = value;
>                 Raise();
>             }
>         }
>
>         public event PropertyChangedEventHandler PropertyChanged;
>         private void Raise([CallerMemberName] string name = null) =>
>             PropertyChanged?.Invoke(this, new PropertyChangedEventArgs(name));
>     }
> }
> ```

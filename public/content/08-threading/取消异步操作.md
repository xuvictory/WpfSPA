> [!example] 完整示例
> **取消异步操作演示：用 CancellationTokenSource 取消批量采集：**
>
> **MainWindow.xaml：**
> ```xml
> <Window x:Class="HmiDemo.MainWindow"
>         xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"
>         xmlns:x="http://schemas.microsoft.com/winfx/2006/xaml"
>         Title="取消异步操作" Height="380" Width="440"
>         WindowStartupLocation="CenterScreen" Background="#0D1117">
>     <StackPanel Margin="15" Background="#161B22" Padding="15">
>         <TextBlock Text="取消异步操作演示" FontSize="16" FontWeight="Bold"
>                    Foreground="#58A6FF" Margin="0,0,0,10"/>
>         <!-- 启动批量采集（共 10 轮，每轮 0.5 秒） -->
>         <Button x:Name="StartButton" Content="启动批量采集（10 轮）" Click="OnStartClick"
>                 Margin="0,5" Padding="8" Background="#238636" Foreground="White"/>
>         <!-- 点击后发出取消请求 -->
>         <Button Content="取消采集" Click="OnCancelClick"
>                 Margin="0,5" Padding="8" Background="#DA3633" Foreground="White"/>
>         <TextBlock x:Name="LogText" Foreground="#8B949E" TextWrapping="Wrap"
>                    MinHeight="150" Margin="0,10,0,0"/>
>     </StackPanel>
> </Window>
> ```
>
> **MainWindow.xaml.cs —— 后台代码：**
> ```csharp
> using System;
> using System.Threading;
> using System.Threading.Tasks;
> using System.Windows;
>
> namespace HmiDemo
> {
>     public partial class MainWindow : Window
>     {
>         private CancellationTokenSource _cts; // 取消令牌源
>
>         public MainWindow() => InitializeComponent();
>
>         private async void OnStartClick(object sender, RoutedEventArgs e)
>         {
>             _cts = new CancellationTokenSource();
>             var token = _cts.Token; // 取出令牌传给任务
>             StartButton.IsEnabled = false;
>             LogText.Text = "";
>             try
>             {
>                 for (int i = 1; i <= 10; i++)
>                 {
>                     token.ThrowIfCancellationRequested(); // 主动检查取消
>                     await Task.Delay(500, token); // 等待期间也可被取消
>                     LogText.Text += $"第 {i} 轮采集完成（通道 {i}）\r\n";
>                 }
>                 LogText.Text += "全部 10 轮采集完成\r\n";
>             }
>             catch (OperationCanceledException)
>             {
>                 LogText.Text += "采集已被用户取消 ✓\r\n";
>             }
>             finally
>             {
>                 StartButton.IsEnabled = true;
>                 _cts?.Dispose();
>             }
>         }
>
>         // 请求取消：任务在下一个检查点响应
>         private void OnCancelClick(object sender, RoutedEventArgs e)
>         {
>             _cts?.Cancel();
>         }
>     }
> }
> ```
>

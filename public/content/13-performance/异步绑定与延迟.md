> [!example] 完整示例
> **异步加载演示：同步加载阻塞 UI vs async/await 异步加载不卡界面：**
>
> **MainWindow.xaml：**
> ```xml
> <Window x:Class="HmiDemo.MainWindow"
>         xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"
>         xmlns:x="http://schemas.microsoft.com/winfx/2006/xaml"
>         Title="异步绑定与延迟" Height="380" Width="520"
>         WindowStartupLocation="CenterScreen" Background="#0D1117">
>     <Grid Margin="15" Background="#161B22" Padding="10">
>         <Grid.RowDefinitions>
>             <RowDefinition Height="Auto"/>
>             <RowDefinition Height="*"/>
>             <RowDefinition Height="Auto"/>
>         </Grid.RowDefinitions>
>         <StackPanel Orientation="Horizontal">
>             <Button Content="同步加载（卡 UI）" Click="OnLoadSync" Padding="8"
>                     Background="#DA3633" Foreground="White"/>
>             <Button Content="异步加载（不卡 UI）" Click="OnLoadAsync" Padding="8" Margin="8,0,0,0"
>                     Background="#238636" Foreground="White"/>
>         </StackPanel>
>         <ListBox x:Name="DataList" Grid.Row="1" Margin="0,10,0,10"
>                  Background="#0D1117" Foreground="#8B949E" BorderBrush="#21262D"/>
>         <TextBlock x:Name="StatusText" Grid.Row="2" Foreground="#58A6FF" TextWrapping="Wrap"/>
>     </Grid>
> </Window>
> ```
>
> **MainWindow.xaml.cs —— 后台代码：**
> ```csharp
> using System;
> using System.Collections.Generic;
> using System.Threading;
> using System.Threading.Tasks;
> using System.Windows;
>
> namespace HmiDemo
> {
>     public partial class MainWindow : Window
>     {
>         public MainWindow() => InitializeComponent();
>
>         // 同步加载：阻塞 UI 线程，窗口无法拖动/响应
>         private void OnLoadSync(object sender, RoutedEventArgs e)
>         {
>             StatusText.Text = "同步加载中…窗口将卡住 2 秒";
>             var data = LoadDataSync();          // 阻塞 2 秒
>             DataList.ItemsSource = data;
>             StatusText.Text = "同步加载完成（期间 UI 无响应）";
>         }
>
>         // 异步加载：await 不阻塞 UI 线程
>         private async void OnLoadAsync(object sender, RoutedEventArgs e)
>         {
>             StatusText.Text = "异步加载中…窗口保持流畅";
>             var data = await LoadDataAsync();   // 后台线程耗时 2 秒
>             DataList.ItemsSource = data;
>             StatusText.Text = "异步加载完成（期间 UI 可正常交互）";
>         }
>
>         private List<string> LoadDataSync()
>         {
>             Thread.Sleep(2000);   // 模拟 2 秒耗时
>             return BuildData();
>         }
>
>         private async Task<List<string>> LoadDataAsync()
>         {
>             // 放到线程池执行，UI 线程不被阻塞，完成后自动切回 UI 线程
>             return await Task.Run(() =>
>             {
>                 Thread.Sleep(2000);
>                 return BuildData();
>             });
>         }
>
>         private List<string> BuildData()
>         {
>             var list = new List<string>();
>             for (int i = 1; i <= 50; i++)
>                 list.Add($"历史记录 {i}：设备 D-{(i % 10) + 1} 数据帧 #{DateTime.Now:HH:mm:ss}");
>             return list;
>         }
>     }
> }
> ```

> [!best] 最佳实践
> - 编写代码时保持一致的命名规范（PascalCase 用于公共成员，_camelCase 用于私有字段）
> - 善用 Visual Studio 的智能提示和代码片段，提高开发效率
> - 每个关键代码块加上注释，解释"为什么这样写"而不仅仅是"写的是什么"
> - 遵循 SOLID 原则，尤其是单一职责原则：一个类只做一件事
> - 经常重构：写完功能后回头看看有没有更简洁的写法

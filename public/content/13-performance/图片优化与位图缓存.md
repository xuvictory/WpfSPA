> **位图加载优化：DecodePixelWidth 缩小解码降低内存 + Freeze 冻结缓存复用：**

> **MainWindow.xaml.cs —— 后台代码：**
> ```csharp
> using System;
> using System.Diagnostics;
> using System.IO;
> using System.Windows;
> using System.Windows.Media;
> using System.Windows.Media.Imaging;
>
> namespace HmiDemo
> {
>     public partial class MainWindow : Window
>     {
>         private BitmapSource _cached;   // 缓存冻结后的位图
>
>         public MainWindow() => InitializeComponent();
>
>         // 生成一张 4000×3000 的模拟大图（约 48 MB 像素数据）
>         private BitmapSource CreateBigImage()
>         {
>             var pixels = new byte[4000 * 3000 * 4];
>             new Random(1).NextBytes(pixels);
>             return BitmapSource.Create(4000, 3000, 96, 96, PixelFormats.Bgra32, null, pixels, 4000 * 4);
>         }
>
>         // 未优化：内存中保留 4000×3000 完整像素，占用大
>         private void OnFullDecode(object sender, RoutedEventArgs e)
>         {
>             var sw = Stopwatch.StartNew();
>             var img = BitmapFrame.Create(CreateBigImage());
>             img.Freeze();
>             DemoImage.Source = img;
>             sw.Stop();
>             ResultText.Text = $"全尺寸解码：保留 4000×3000 像素，耗时 {sw.Elapsed.TotalMilliseconds:F0} ms，内存占用大";
>         }
>
>         // 优化：DecodePixelWidth 只按显示所需尺寸解码，显著降低内存
>         private void OnDecodePixelWidth(object sender, RoutedEventArgs e)
>         {
>             var sw = Stopwatch.StartNew();
>             var big = CreateBigImage();
>             var encoder = new PngBitmapEncoder();
>             encoder.Frames.Add(BitmapFrame.Create(big));
>             using (var ms = new MemoryStream())
>             {
>                 encoder.Save(ms);
>                 ms.Position = 0;
>                 var img = new BitmapImage();
>                 img.BeginInit();
>                 img.DecodePixelWidth = 400;          // 只解码到 400 像素宽
>                 img.CacheOption = BitmapCacheOption.OnLoad;
>                 img.StreamSource = ms;
>                 img.EndInit();
>                 img.Freeze();                        // 冻结后可用于跨线程缓存共享
>                 DemoImage.Source = img;
>             }
>             sw.Stop();
>             ResultText.Text = $"DecodePixelWidth=400：解码到 400 像素宽，耗时 {sw.Elapsed.TotalMilliseconds:F0} ms，内存骤降";
>         }
>
>         // 缓存：首次创建后 Freeze 并复用，后续加载零开销
>         private void OnUseCache(object sender, RoutedEventArgs e)
>         {
>             if (_cached == null)
>             {
>                 var sw = Stopwatch.StartNew();
>                 _cached = CreateBigImage();
>                 _cached.Freeze();
>                 sw.Stop();
>                 CacheInfoText.Text = $"首次创建并冻结缓存，耗时 {sw.Elapsed.TotalMilliseconds:F0} ms";
>             }
>             DemoImage.Source = _cached;
>             ResultText.Text = "已从缓存复用位图，后续加载零开销";
>         }
>     }
> }
> ```

> [!example] 完整示例
> **海量图元可视化：2000 个 UI 控件 vs DrawingVisual 一次性绘制 5 万个点：**
>
> **MainWindow.xaml：**
> ```xml
> <Window x:Class="HmiDemo.MainWindow"
>         xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"
>         xmlns:x="http://schemas.microsoft.com/winfx/2006/xaml"
>         Title="大量控件同时可见" Height="430" Width="560"
>         WindowStartupLocation="CenterScreen" Background="#0D1117">
>     <StackPanel Margin="15" Background="#161B22" Padding="15">
>         <TextBlock Text="海量数据可视化：控件 vs DrawingVisual"
>                    Foreground="#58A6FF" FontSize="16" FontWeight="Bold"/>
>         <Border Background="#0D1117" BorderBrush="#21262D" BorderThickness="1"
>                 Height="220" Margin="0,10,0,0">
>             <Grid>
>                 <Canvas x:Name="DrawCanvas" Background="Transparent"/>
>                 <TextBlock Text="点击下方按钮生成海量图元" Foreground="#8B949E"
>                            Margin="8" VerticalAlignment="Top"/>
>             </Grid>
>         </Border>
>         <TextBlock x:Name="ResultText" Foreground="#8B949E" Margin="0,10,0,0" TextWrapping="Wrap"/>
>         <StackPanel Orientation="Horizontal" Margin="0,10,0,0">
>             <Button Content="添加 2000 个控件（慢）" Click="OnAddControls" Padding="8"
>                     Background="#21262D" Foreground="White"/>
>             <Button Content="绘制 5 万点 DrawingVisual（快）" Click="OnDrawVisual" Padding="8" Margin="8,0,0,0"
>                     Background="#58A6FF" Foreground="White"/>
>             <Button Content="清空" Click="OnClear" Padding="8" Margin="8,0,0,0"
>                     Background="#21262D" Foreground="White"/>
>         </StackPanel>
>     </StackPanel>
> </Window>
> ```
>
> **MainWindow.xaml.cs —— 后台代码：**
> ```csharp
> using System;
> using System.Collections.Generic;
> using System.Diagnostics;
> using System.Windows;
> using System.Windows.Controls;
> using System.Windows.Media;
>
> namespace HmiDemo
> {
>     public partial class MainWindow : Window
>     {
>         private readonly VisualHost _host = new VisualHost();  // 承载 DrawingVisual 的宿主
>         private int _controlCount;
>
>         public MainWindow()
>         {
>             InitializeComponent();
>             DrawCanvas.Children.Add(_host);
>         }
>
>         // 低效做法：为每个数据点创建一个 UIElement，控件过多会拖垮布局与渲染
>         private void OnAddControls(object sender, RoutedEventArgs e)
>         {
>             var sw = Stopwatch.StartNew();
>             var rnd = new Random(42);
>             for (int i = 0; i < 2000; i++)
>             {
>                 var dot = new Border
>                 {
>                     Width = 2,
>                     Height = 2,
>                     Background = new SolidColorBrush(Color.FromRgb(0x8B, 0x94, 0x9E)),
>                     HorizontalAlignment = HorizontalAlignment.Left,
>                     VerticalAlignment = VerticalAlignment.Top,
>                     Margin = new Thickness(rnd.Next(0, (int)DrawCanvas.ActualWidth),
>                                            rnd.Next(0, (int)DrawCanvas.ActualHeight), 0, 0)
>                 };
>                 DrawCanvas.Children.Add(dot);
>             }
>             sw.Stop();
>             _controlCount += 2000;
>             ResultText.Text = $"已添加 {_controlCount} 个控件，耗时 {sw.Elapsed.TotalMilliseconds:F0} ms" +
>                               "（控件越多，布局与渲染越慢）";
>         }
>
>         // 高效做法：用 DrawingVisual 一次性绘制，不产生任何 UIElement
>         private void OnDrawVisual(object sender, RoutedEventArgs e)
>         {
>             var sw = Stopwatch.StartNew();
>             var visual = new DrawingVisual();
>             using (var dc = visual.RenderOpen())
>             {
>                 var brush = new SolidColorBrush(Color.FromRgb(0x58, 0xA6, 0xFF));
>                 brush.Freeze();   // 冻结后可跨线程共享，减少开销
>                 var rnd = new Random(42);
>                 for (int i = 0; i < 50000; i++)
>                 {
>                     var x = rnd.Next(0, (int)DrawCanvas.ActualWidth);
>                     var y = rnd.Next(0, (int)DrawCanvas.ActualHeight);
>                     dc.DrawRectangle(brush, null, new Rect(x, y, 2, 2));
>                 }
>             }
>             _host.Add(visual);
>             sw.Stop();
>             ResultText.Text = $"DrawingVisual 绘制 50000 个点，耗时 {sw.Elapsed.TotalMilliseconds:F1} ms，" +
>                               "界面只有 1 个 Visual 节点，控件数不变";
>         }
>
>         private void OnClear(object sender, RoutedEventArgs e)
>         {
>             _host.Clear();
>             _controlCount = 0;
>             ResultText.Text = "已清空";
>         }
>     }
>
>     // 自定义视觉宿主：负责把 DrawingVisual 挂进可视化树
>     internal class VisualHost : FrameworkElement
>     {
>         private readonly List<Visual> _visuals = new List<Visual>();
>
>         protected override int VisualChildrenCount => _visuals.Count;
>
>         protected override Visual GetVisualChild(int index) => _visuals[index];
>
>         public void Add(Visual v)
>         {
>             _visuals.Add(v);
>             AddVisualChild(v);
>         }
>
>         public void Clear()
>         {
>             foreach (var v in _visuals) RemoveVisualChild(v);
>             _visuals.Clear();
>         }
>     }
> }
> ```
